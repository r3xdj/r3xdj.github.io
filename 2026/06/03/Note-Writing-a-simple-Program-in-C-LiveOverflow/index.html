C 語言

#include <stdio.h>
int main(int argc, char *argv[]){
    if(argc==2){
        printf("Knock, Knock, %s\n", argv[1]);
    } else {
        fprintf(stderr, "Usage: %s <name>\n", argv[0]);
        return 1;
    }
}

解析：

  • argc: 參數數量(從檔案名稱算起)

  • argv: 參數矩陣(包含檔案名稱)
    範例：

    #include <stdio.h>
int main(int argc, char *argv[]){
  printf("argc: %d\n",argc);
  printf("argv:\n")
  for(int i=0;i<argc;i++){
      printf("\targv[%d]: %s\n", i, argv[i]);
  }
  return 0;
}
